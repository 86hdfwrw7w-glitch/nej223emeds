# dokument-review — Technische Referenz

Skripte sind **config-getrieben**: Claude schreibt eine `config.json` in den Scratch-Ordner und ruft jedes Skript als `python <skript> <config.json>` (mit `PYTHONIOENCODING=utf-8`). Pfade in `config.json` mit doppelten Backslashes oder Forward-Slashes.

`config.json`-Felder (über die Schritte hinweg ergänzt):
```json
{
  "pdf_path": "…/Arbeit.pdf",
  "out": "…/scratchpad/dokument-review",
  "sections": {"00_frontmatter": [1,8], "01_einleitung": [9,10], "…": [11,19]},
  "biblio_pages": [72,76],
  "body_pages": [9,71],
  "papers_dir": "…/alle paper/Paper nach Themen",
  "task_output": "…/tasks/<id>.output",
  "dest": "…/Review-Report_<stem>_<date>.md"
}
```

## extract.py  (Schritt 1)
```python
# -*- coding: utf-8 -*-
import fitz, json, sys, os
cfg = json.load(open(sys.argv[1], encoding="utf-8"))
out = cfg["out"]; os.makedirs(out, exist_ok=True)
doc = fitz.open(cfg["pdf_path"])
print("PAGES:", doc.page_count)
parts = []
for i, page in enumerate(doc):
    parts.append(f"\n\n===== PAGE {i+1} =====\n{page.get_text('text')}")
open(os.path.join(out, "thesis_text.txt"), "w", encoding="utf-8").write("".join(parts))
json.dump(doc.get_toc(), open(os.path.join(out, "toc.json"), "w", encoding="utf-8"),
          ensure_ascii=False, indent=2)
print("TOC entries:", len(doc.get_toc()))
print("--- per-page opening lines (zum Bestimmen der Kapitelgrenzen) ---")
for i, page in enumerate(doc):
    lines = [l.strip() for l in page.get_text("text").splitlines() if l.strip()]
    print(f"p{i+1}: " + " | ".join(lines[:4])[:120])
```
**Offset bestimmen:** Im Inhaltsverzeichnis steht je Kapitel die *Body*-Seite. Eine davon im Erstzeilen-Dump suchen → physische Seite. `offset = phys - body`. Dann alle Kapitel: `phys = body + offset`. Verifizieren mit einer zweiten Überschrift.

## prep.py  (Schritt 2)
```python
# -*- coding: utf-8 -*-
import json, sys, re, os, glob
cfg = json.load(open(sys.argv[1], encoding="utf-8"))
out = cfg["out"]; SECT = os.path.join(out, "sections"); os.makedirs(SECT, exist_ok=True)
txt = open(os.path.join(out, "thesis_text.txt"), encoding="utf-8").read()
parts = re.split(r"===== PAGE (\d+) =====", txt)
pmap = {int(parts[i]): parts[i+1] for i in range(1, len(parts), 2)}
def slab(a, b): return "".join(f"\n===== PAGE {n} =====\n" + pmap.get(n, "") for n in range(a, b+1))
for name, (a, b) in cfg["sections"].items():
    open(os.path.join(SECT, name + ".txt"), "w", encoding="utf-8").write(slab(a, b))
ba, bb = cfg["biblio_pages"]
open(os.path.join(out, "bibliography.txt"), "w", encoding="utf-8").write(slab(ba, bb))
ya, yb = cfg["body_pages"]
body = "".join(pmap.get(n, "") for n in range(ya, yb+1)).replace("\n", " ")
pat = re.compile(r"([A-ZÄÖÜ][A-Za-zÄÖÜäöüß'’\-]+(?:\s+(?:und|&|et al\.?)\s+[A-ZÄÖÜ][A-Za-zÄÖÜäöüß'’\-]+)*)\s*\((\d{4}[a-z]?)\)")
cites = sorted(set(f"{m[0]} ({m[1]})" for m in pat.findall(body)))
open(os.path.join(out, "citations_narrative.txt"), "w", encoding="utf-8").write("\n".join(cites))
if cfg.get("papers_dir"):
    papers = sorted(glob.glob(os.path.join(cfg["papers_dir"], "**", "*.pdf"), recursive=True))
    open(os.path.join(out, "papers_inventory.txt"), "w", encoding="utf-8").write("\n".join(papers))
    print("papers:", len(papers))
print("sections:", list(cfg["sections"]), "| narrative citations:", len(cites))
```

## render.py  (Schritt 5)
Liest die Workflow-Output-Datei (Wrapper → `result`), rendert den Report. Lässt eine Marke `<!-- GESAMTEINSCHAETZUNG -->` für die handgeschriebene Einschätzung (Claude ersetzt sie per Edit in Schritt 6).
```python
# -*- coding: utf-8 -*-
import json, sys, re, html
cfg = json.load(open(sys.argv[1], encoding="utf-8"))
data = json.load(open(cfg["task_output"], encoding="utf-8"))["result"]
U = lambda x: html.unescape(x) if isinstance(x, str) else x
flat = [{k: U(v) for k, v in f.items()} for f in data["flat"]]
ORDER = [U(b) for b in data["ORDER"]]
counts = data["counts"]; byBucket = {U(k): v for k, v in data["byBucket"].items()}; byKind = data["byKind"]
one = lambda t: re.sub(r"\s+", " ", (t or "").strip())
SEV = {"kritisch": 0, "wichtig": 1, "kosmetisch": 2}
BADGE = {"kritisch": "🔴 KRITISCH", "wichtig": "🟠 WICHTIG", "kosmetisch": "🟡 Kosmetik"}
KIND_ORDER = ["Zitat-Inhalt","Inhalt/Argumentation","Zahl/Daten","Notebook-Logik","Querverweis",
 "Konsistenz","Formatierung/Notation","Notation","Zitat-Format","Quelle-Web",
 "Grammatik/Rechtschreibung","Rechtschreibung","Zeichensetzung","Stil/Ausdruck","Abkürzungsverzeichnis","Sonstiges"]
krank = lambda k: KIND_ORDER.index(k) if k in KIND_ORDER else len(KIND_ORDER)
L = []; A = L.append
A(f"# Review-Report — {cfg.get('title','Dokument-Review')}")
A("")
A(f"**Basis:** `{cfg.get('pdf_name','')}` · **Geprüft am:** {cfg.get('date','')} · **Quelldokument unverändert.**  ")
A(f"**Befunde gesamt: {counts['total']}** — 🔴 {counts['kritisch']} kritisch · 🟠 {counts['wichtig']} wichtig · 🟡 {counts['kosmetisch']} kosmetisch.")
A(""); A("<!-- GESAMTEINSCHAETZUNG -->"); A("")
A("## Befundübersicht"); A(""); A("| Abschnitt | Befunde |"); A("|---|---:|")
for b in ORDER:
    if b in byBucket: A(f"| {b} | {byBucket[b]} |")
A(f"| **Gesamt** | **{counts['total']}** |"); A("")
A("| Typ | Anzahl |"); A("|---|---:|")
for k, v in sorted(byKind.items(), key=lambda x: -x[1]): A(f"| {U(k)} | {v} |")
A(""); A("---"); A(""); A("## 🔴 Kritische Befunde zuerst"); A("")
krit = [f for f in flat if f["severity"] == "kritisch"]
if not krit: A("_Keine kritischen Befunde._")
for f in krit:
    loc = ("S." + f["page"]) if f.get("page") else "—"
    A(f"- **[{f['bucket']} · {loc}]** {one(f['problem'])}" + (f" · Quelle: {f['cite']}" if f.get('cite') else ""))
    if f.get("quote"): A(f"  - _Zitat:_ „{one(f['quote'])}“")
    if f.get("suggestion"): A(f"  - _Vorschlag:_ {one(f['suggestion'])}")
A(""); A("---"); A(""); A("## Befunde im Detail (nach Kapitel & Typ)"); A("")
groups = {}
for f in flat: groups.setdefault(f["bucket"], []).append(f)
for b in ORDER:
    if b not in groups: continue
    items = groups[b]
    nk = sum(x["severity"]=="kritisch" for x in items); nw = sum(x["severity"]=="wichtig" for x in items); nc = sum(x["severity"]=="kosmetisch" for x in items)
    A(f"### {b}  ({len(items)}: {nk} kritisch · {nw} wichtig · {nc} kosmetisch)"); A("")
    for k in sorted(set(x["kind"] for x in items), key=krank):
        sub = sorted([x for x in items if x["kind"]==k], key=lambda x: SEV.get(x["severity"],9))
        A(f"**{U(k)}** ({len(sub)})"); A("")
        for f in sub:
            loc = ("S." + f["page"]) if f.get("page") else "—"
            cite = (f" · _{f['cite']}_") if f.get("cite") else ""
            A(f"- **{BADGE.get(f['severity'],f['severity'])} · {loc}**{cite} · Konfidenz: {f.get('confidence','')}  ")
            A(f"  {one(f['problem'])}  ")
            if f.get("quote"): A(f"  _Zitat:_ „{one(f['quote'])}“  ")
            if f.get("suggestion"): A(f"  → **Vorschlag:** {one(f['suggestion'])}")
        A("")
    A("---"); A("")
open(cfg["dest"], "w", encoding="utf-8").write("\n".join(L))
print("WROTE", cfg["dest"], len("\n".join(L)), "chars")
```

## Workflow-Vorlage (Schritt 4)

Inline an das `Workflow`-Tool übergeben. **ANPASSEN**: alle `SCRATCH`/Pfade, `CHAPTERS`, `PAPER_MAP`, `NOTEBOOKS`, `TABLES`, `bucketForPage`-Grenzen. **Rückgabe bleibt `{counts,byBucket,byKind,ORDER,flat,notes}`** — kein Agent-Render.

```javascript
export const meta = {
  name: 'dokument-review-run',
  description: 'Multi-Agenten-Review einer PDF: Lektorat, Quellen, Zahlen/Notebooks, Querschnitt; gibt volle Befunddaten zurück',
  phases: [{title:'Kapitel-Lektorat'},{title:'Quellen'},{title:'Zahlen-Notebooks'},{title:'Querschnitt'}],
}
const SCRATCH = "…\\scratchpad\\dokument-review"        // ANPASSEN
const SECT = SCRATCH + "\\sections"
const THESIS_TXT = SCRATCH + "\\thesis_text.txt"
const BIB = SCRATCH + "\\bibliography.txt"
const PAPERS = "…\\Paper nach Themen"                    // ANPASSEN (papers_dir)
const TABLES = "…\\tables"                               // ANPASSEN
const NBDIR  = "…\\Notebooks"                            // ANPASSEN

const F = { type:"object", properties:{ findings:{type:"array", items:{ type:"object", properties:{
  page:{type:"string",description:"Physische Seitenzahl als reine Zahl; leer wenn n/a"},
  kind:{type:"string",description:"Grammatik/Rechtschreibung|Zeichensetzung|Stil/Ausdruck|Formatierung/Notation|Inhalt/Argumentation|Zitat-Inhalt|Zitat-Format|Quelle-Web|Querverweis|Zahl/Daten|Notebook-Logik|Konsistenz|Sonstiges"},
  severity:{type:"string",enum:["kritisch","wichtig","kosmetisch"]},
  quote:{type:"string"}, problem:{type:"string"}, suggestion:{type:"string"},
  confidence:{type:"string",enum:["hoch","mittel","niedrig"]} },
  required:["kind","severity","problem","suggestion"] }}, note:{type:"string"} }, required:["findings"] }

const RULES = `Du prüfst eine wissenschaftliche Arbeit. Arbeite NUR an deiner zugewiesenen Quelle, erfinde nichts (unsicher -> confidence niedrig). Zitate wortwörtlich + kurz. Seitenzahl = Zahl hinter "===== PAGE N". Echte Umlaute. severity: kritisch=inhaltlich falsch/falsche Quelle/falsche Zahl; wichtig=klarer Fehler/Schwäche; kosmetisch=Feinschliff. Keine Dubletten.`

// --- Phase 1: Kapitel ---  (CHAPTERS ANPASSEN: key, file, info)
const CHAPTERS = [
  {key:"00 Front-Matter", file:"00_frontmatter.txt", info:"Titel, Erklärung, Verzeichnisse (phys. 1-8)"},
  {key:"01 Einleitung", file:"01_einleitung.txt", info:"Kap.1 (phys. …)"},
  // … weitere Kapitel + {key:"09 Anhang", file:"10_anhang.txt", info:"…"}
]
phase('Kapitel-Lektorat')
const chapterResults = await parallel(CHAPTERS.map((c)=>()=>
  agent(`${RULES}\n\nAUFGABE: Vollständiges Lektorat "${c.key}" — ${c.info}. Lies ${SECT}\\${c.file}. Prüfe: Rechtschreibung/Grammatik/Zeichensetzung; Stil/Ausdruck; Formatierung/Notation (einheitliche Schreibweisen, Dezimalkomma, Abkürzungseinführung); Inhalt/Argumentation (Logik, unbelegte Behauptungen, Widersprüche); In-Text-Zitate (Form + belegt + kommentiert). Front-Matter zusätzlich: Pflichtbestandteile, Verzeichnis-Konsistenz, Abkürzungsverzeichnis vollständig. findings mit physischer Seite.`,
    {label:`lektorat:${c.key}`, phase:'Kapitel-Lektorat', schema:F})
    .then((r)=>({track:"chapter", bucketDefault:c.key, data:r}))))

// --- Phase 2: Quellen ---
phase('Quellen')
const bibConsistency = agent(`${RULES}\n\nAUFGABE: Literaturverzeichnis + Beleg-Konsistenz (kein PDF-Inhalt). Lies ${BIB}; Grep auf ${THESIS_TXT}. Prüfe: einheitliches Zitierformat (& vs und, S. x–y, doi, Klammern), alphabetische Sortierung, Dubletten, uneinheitliche Namen; jeder In-Text-Beleg hat Eintrag (fehlend=kritisch); jeder Eintrag wird zitiert (Pseudo-Beleg=wichtig, per Grep des Nachnamens); Jahres-/Namenskonflikte Text vs. Verzeichnis. kind Zitat-Format/Konsistenz.`,
  {label:'quellen:bib', phase:'Quellen', schema:F}).then((r)=>({track:"bib", bucketDefault:"10 Literaturverzeichnis & Quellen", data:r}))

// PAPER_MAP ANPASSEN: [relPathUnterPAPERS, "Cite (Jahr)", grepKeyNachname, "s"|"m"]  (s=Substanz, m=Methode)
const PAPER_MAP = [ /* ["02_…\\Autor_Jahr.pdf","Autor & X (Jahr)","Autor","s"], … */ ]
const paperChecks = PAPER_MAP.map(([rel,cite,grepKey,kind])=>{
  const sub = kind==="s"
    ? `SUBSTANZ-Quelle: Prüfe, ob ALLE Aussagen, die die Arbeit dieser Quelle zuschreibt (Ergebnisse, Methoden, Länder/Zeiträume, Richtung/Größe von Effekten), getreu wiedergegeben sind (keine Übertreibung/falsche Zuschreibung/verdrehten Ergebnisse).`
    : `METHODEN-Quelle: Prüfe, ob das Verfahren korrekt dieser Quelle zugeschrieben und korrekt beschrieben wird.`
  return ()=>agent(`${RULES}\n\nAUFGABE: Inhaltlicher Quellen-Abgleich für ${cite}. 1) Grep ${THESIS_TXT} nach "${grepKey}" (richtiges Jahr beachten), notiere physische Seite. 2) Lies ${PAPERS}\\${rel} (>10 S.: gezielt Abstract/Intro/Ergebnisse mit pages). 3) ${sub} 4) Verzeichnis-Metadaten (Titel/Jahr/Journal/Band/Seiten) gegen das echte Paper prüfen -> kind Zitat-Format. Korrekte Wiedergabe: findings:[] + note "korrekt". Inhaltliche Abweichung: kind Zitat-Inhalt, page=Seite der Textstelle.`,
    {label:`paper:${grepKey}`, phase:'Quellen', schema:F}).then((r)=>({track:"paper", cite, bucketDefault:"10 Literaturverzeichnis & Quellen", data:r}))})

// WEB_BATCHES ANPASSEN: Quellen ohne lokales PDF (Klassiker/Lehrbücher) + Online-/Institutionsquellen, in Gruppen
const WEB_BATCHES = [ /* "Eintrag 1; Eintrag 2; …", … */ ]
const webChecks = WEB_BATCHES.map((entries,i)=>()=>agent(`${RULES}\n\nAUFGABE: Web-Verifikation: ${entries}. Lade Tools: ToolSearch "select:WebSearch,WebFetch". Prüfe je Quelle: Existenz; Autor/Jahr/Titel/Journal/Band/Seiten/DOI; bei URLs Erreichbarkeit + Passung. Verzeichniszeilen in ${BIB}. Abweichungen als kind Quelle-Web; Korrektes nur in note.`,
  {label:`web:${i+1}`, phase:'Quellen', schema:F}).then((r)=>({track:"web", bucketDefault:"10 Literaturverzeichnis & Quellen", data:r})))
const sourceResults = await parallel([()=>bibConsistency, ...paperChecks, ...webChecks])

// --- Phase 3: Zahlen + Notebooks ---  (TABLES/NOTEBOOKS + Tabellenzuordnung ANPASSEN)
phase('Zahlen-Notebooks')
const numbersMain = ()=>agent(`${RULES}\n\nAUFGABE: Zahlen-Abgleich Ergebnis-Kapitel Text vs. Output-CSVs. Lies das Ergebnis-Abschnittsfile und die relevanten CSVs in ${TABLES}. Gleiche JEDE Zahl (RMSE/MAE/Ratios/Koeffizienten/SE/p-Werte/n/R²) ab. Achte auf Rundung, Dezimalkomma vs. Punkt, vertauschte Spalten, Vorzeichen, Übertragungsfehler, Text-vs-Tabelle-Inkonsistenz. kind Zahl/Daten, page=Seite.`,
  {label:'zahlen:ergebnisse', phase:'Zahlen-Notebooks', schema:F}).then((r)=>({track:"numbers", bucketDefault:"06 Empirische Ergebnisse", data:r}))
const numbersData = ()=>agent(`${RULES}\n\nAUFGABE: Zahlen-Abgleich Daten/Anhang Text vs. CSVs in ${TABLES}. Lies Daten- + Anhang-Abschnittsfile. Stationaritäts-Klassifikation, Korrelationen (Werte/Vorzeichen/Lags), deskriptive Kennzahlen, Test-p-Werte abgleichen. kind Zahl/Daten, page=Seite.`,
  {label:'zahlen:daten', phase:'Zahlen-Notebooks', schema:F}).then((r)=>({track:"numbers", bucketDefault:"05 Daten", data:r}))
const NOTEBOOKS = [ /* ["01_Download.ipynb","Kurzbeschreibung"], … */ ]
const nbChecks = NOTEBOOKS.map(([file,desc])=>()=>agent(`${RULES}\n\nAUFGABE: Logik-/Methodik-Review Notebook "${file}" (${desc}). Lies ${NBDIR}\\${file}; zur Einordnung das Methodik-Abschnittsfile. Prüfe: Methodik-Umsetzung (Stationaritätstests, Differenzierung, Standardisierung, Rolling-Window ohne Look-ahead/Leakage, direkte vs. iterative Mehrschritt, HAC-Lag, Clark-West vs. Diebold-Mariano, RMSE/MAE); Programmierfehler (Split-Lecks, Lag-Indexverschiebung, NaN, falsche Spalten, hartkodierte Werte, inkonsistente n); Konsistenz Code vs. Arbeit. kind Notebook-Logik, page leer. note=1-2 Sätze, was das Notebook tut.`,
  {label:`nb:${file.slice(0,12)}`, phase:'Zahlen-Notebooks', schema:F}).then((r)=>({track:"notebook", file, bucketDefault:"11 Notebooks & Reproduzierbarkeit", data:r})))
const znResults = await parallel([numbersMain, numbersData, ...nbChecks])
const nbNotes = znResults.filter((x)=>x&&x.track==="notebook").map((x)=>`- ${x.file}: ${(x.data&&x.data.note)||""}`).join("\n")
const nbSynth = await agent(`${RULES}\n\nAUFGABE: Gesamt-Kohärenz der Pipeline (Notebooks) + Abgleich mit den Kernaussagen der Arbeit. Kurzfassungen:\n${nbNotes}\nPrüfe durchgängige Konsistenz von Stichprobenzeitraum/n/Fenster, ob die berichteten Ergebnisse durch die Pipeline gestützt werden, methodische Brüche zwischen Notebooks. kind Notebook-Logik/Konsistenz.`,
  {label:'nb:synthese', phase:'Zahlen-Notebooks', schema:F}).then((r)=>({track:"notebook", bucketDefault:"11 Notebooks & Reproduzierbarkeit", data:r}))

// --- Phase 4: Querschnitt ---

Jede "Abbildung/Tabelle/Kapitel N"-Referenz existiert + richtige Nummer/Titel; keine verwaisten Objekte; Verzeichnis-Einträge stimmen mit Beschriftungen; Nummerierungslücken. kind Querverweis, page=Seite.`,
  {label:'quer:xref', phase:'Querschnitt', schema:F}).then((r)=>({track:"xref", bucketDefault:"12 Querschnitt", data:r}))
const redthread = ()=>agent(`${RULES}\n\nAUFGABE: Roter Faden / Forschungsfrage / Hypothesen. Lies Einleitung-, Methodik-, Ergebnis-, Diskussion-, Fazit-Abschnittsfiles. Forschungsfrage klar + im Fazit beantwortet? Hypothesen konsistent abgeleitet/bewertet/aufgegriffen? Einleitung↔Fazit bezogen, keine neuen Behauptungen im Fazit? Grenzen transparent, Ergebnisse sachlich? kind Inhalt/Argumentation oder Konsistenz, page=Seite.`,
  {label:'quer:rot', phase:'Querschnitt', schema:F}).then((r)=>({track:"redthread", bucketDefault:"12 Querschnitt", data:r}))
const terminology = ()=>agent(`${RULES}\n\nAUFGABE: Terminologie-/Notations-/Abkürzungs-Konsistenz. Lies Abkürzungsverzeichnis (Front-Matter-File) + Grep ${THESIS_TXT}. Jede genutzte Abkürzung gelistet (und umgekehrt keine ungenutzten)? Einheitliche Symbol-/Modell-/Kennzahl-Notation? Einheitliche Eigennamen + Dezimal-/Zeitformat? kind Konsistenz/Formatierung/Notation, page=Seite.`,
  {label:'quer:term', phase:'Querschnitt', schema:F}).then((r)=>({track:"terminology", bucketDefault:"12 Querschnitt", data:r}))
const querResults = await parallel([xref, redthread, terminology])

// --- Aggregation ---  (bucketForPage-Grenzen ANPASSEN an die Kapitel-Seitenranges)
function bucketForPage(pStr){ const p=parseInt(String(pStr).replace(/[^0-9]/g,""),10); if(!p)return null
  if(p<=8)return "00 Front-Matter"; if(p<=10)return "01 Einleitung"; /* … */ if(p<=76)return "10 Literaturverzeichnis & Quellen"; return "09 Anhang" }
const all = [...chapterResults, ...sourceResults, ...znResults, nbSynth, ...querResults].filter(Boolean)
const flat = []
for (const r of all) for (const f of ((r.data&&r.data.findings)||[])) {
  const byPage = bucketForPage(f.page); let bucket = r.bucketDefault
  if (byPage && ["chapter","numbers","xref","redthread","terminology"].includes(r.track)) bucket = byPage
  if (r.track==="paper" && byPage && f.kind!=="Zitat-Format") bucket = byPage
  flat.push({bucket, track:r.track, cite:r.cite||"", file:r.file||"", page:f.page||"",
    kind:f.kind||"Sonstiges", severity:f.severity||"wichtig", quote:f.quote||"",
    problem:f.problem||"", suggestion:f.suggestion||"", confidence:f.confidence||"mittel"})
}
const ORDER = ["00 Front-Matter","01 Einleitung","02 Literaturüberblick","03 Theorie","04 Methodik","05 Daten","06 Empirische Ergebnisse","07 Diskussion","08 Fazit","09 Anhang","10 Literaturverzeichnis & Quellen","11 Notebooks & Reproduzierbarkeit","12 Querschnitt"]
const groups = {}; for (const f of flat){ (groups[f.bucket]=groups[f.bucket]||[]).push(f) }
const counts = {total:flat.length, kritisch:flat.filter(f=>f.severity==="kritisch").length, wichtig:flat.filter(f=>f.severity==="wichtig").length, kosmetisch:flat.filter(f=>f.severity==="kosmetisch").length}
const byBucket = {}; for (const b of ORDER) if (groups[b]) byBucket[b]=groups[b].length
const byKind = {}; for (const f of flat) byKind[f.kind]=(byKind[f.kind]||0)+1
const notes = all.filter(r=>r.data&&r.data.note).map(r=>({track:r.track, cite:r.cite||r.file||"", note:r.data.note}))
log(`Aggregiert: ${flat.length} Befunde aus ${all.length} Agenten`)
return { counts, byBucket, byKind, ORDER, flat, notes }
```

## Lessons Learned (aus dem ersten Lauf)

- **Render-Agenten sind unzuverlässig** bei großen Läufen (API „Connection closed mid-response“). Deshalb gibt der Workflow Daten zurück und `render.py` rendert lokal. Falls ein *Analyse*-Agent ausfällt: Skript leicht editieren und `Workflow({scriptPath, resumeFromRunId})` — gecachte Agenten kommen sofort.
- **Output-Datei ist ein Wrapper**: `{summary, agentCount, logs, result}` — Befunde unter `result`. Bucket-Namen können HTML-escaped sein (`&amp;`) → `html.unescape`.
- **Windows-Konsole** bricht bei Unicode (z. B. `𝑚`, `Δ`) → immer `PYTHONIOENCODING=utf-8`.
- **Pandas nicht nötig** — CSV als Text lesen reicht den Agenten.
- **Seiten-Offset** (römisches Front-Matter) verschiebt alles; immer gegen eine echte Überschrift verifizieren.
- **Concurrency** ist auf ~16 gedeckelt; viele Quellen-Agenten laufen in mehreren Wellen — eingeplant.
- Erster Lauf zum Kalibrieren: 81-Seiten-Thesis, 37 lokale Quellen → 77 Agenten, ~3,5 Mio. Tokens, ~24 Min, 246 Befunde.
